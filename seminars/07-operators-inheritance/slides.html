#include <iostream>

class Ellipse {
  public:
    void SetSize(double width, double height) {
        width_ = width;
        height_ = height;
    }

    double Width() const {
        return width_;
    }

    double Height() const {
        return height_;
    }

  private:
    double width_ = 1;
    double height_ = 1;
};

class Circle : public Ellipse {
  public:
    double Radius() const {
        return Width() / 2;
    }
};

int main() {
    Circle c;
    c.SetSize(2, 6);
    std::cout << c.Radius() << ' ' << c.Width() << ' ' << c.Height() << '\n';
