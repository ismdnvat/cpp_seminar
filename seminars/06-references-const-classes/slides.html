Что напечатает программа?

#include <iostream>

class Account {
  public:
    int GetId() const {
        return id_;
    }

  private:
    int id_ = 7;
    int balance_ = 100;
};

struct AccountView {
    int id;
    int balance;
};

int main() {
    Account account;
    AccountView& view = reinterpret_cast<AccountView&>(account);
    std::cout << view.balance << '\n';
    view.balance = 1000000;
    std::cout << view.balance << '\n';
}
