Less и PrintReversed

#include <cassert>
#include <iostream>
#include <vector>

// 1. Less(int a, unsigned int b) — честное «a < b», и для отрицательных a.
// 2. PrintReversed — печатает вектор задом наперёд. Индекс — size_t, без UB.

int main() {
    assert(Less(-1, 1u));
    assert(Less(3, 5u));
    assert(!Less(5, 3u));
    assert(!Less(7, 7u));
    assert(Less(-2000000000, 4000000000u));

    PrintReversed({1, 2, 3});  // 321
    PrintReversed({});         // пустая строка

    std::cout << "OK\n";
}
  • Напишите Less и PrintReversed, чтобы main напечатал 321, пустую строку и OK.

Что напечатает программа?

#include <iostream>

class Number {
  public:
    Number(int value) : value_(value) {
    }

    Number& operator+=(const Number& other) {
        value_ += other.value_;
        return *this;
    }

    int Value() const {
        return value_;
    }

  private:
    int value_;
};

Number operator+(Number left, const Number& right) {
    return left += right;
}

int main() {
    Number a = 5;
    std::cout << (a + 1).Value() << (1 + a).Value() << '\n';
}
